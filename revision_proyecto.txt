#define _POSIX_C_SOURCE 200809L
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <sys/types.h>
#include "listas.h"

// deja la lista vacía y pone la cantidad de nodos en cero
void inicializar_lista(Lista *lista) {
    lista->inicio = NULL;
    lista->final = NULL;
    lista->cantidad = 0;
}

// crea un nodo con el nombre y lo agrega al final de la lista
int insertar_final(Lista *lista, const char *nombre) {
    Nodo *nuevo = malloc(sizeof(Nodo));

    if (nuevo == NULL) {
        return 0;
    }

    nuevo->nombre = malloc(strlen(nombre) + 1);

    if (nuevo->nombre == NULL) {
        free(nuevo);
        return 0;
    }

    strcpy(nuevo->nombre, nombre);
    nuevo->siguiente = NULL;

    if (lista->inicio == NULL) {
        lista->inicio = nuevo;
    } else {
        lista->final->siguiente = nuevo;
    }

    lista->final = nuevo;
    lista->cantidad++;

    return 1;
}

// recorre los nodos desde el inicio e imprime cada nombre
void imprimir_lista(const Lista *lista) {
    Nodo *actual = lista->inicio;

    while (actual != NULL) {
        printf("%s\n", actual->nombre);
        actual = actual->siguiente;
    }
}

// recorre la lista, libera la memoria y la deja vacía
void liberar_lista(Lista *lista) {
    Nodo *actual = lista->inicio;

    while (actual != NULL) {
        Nodo *siguiente = actual->siguiente;

        free(actual->nombre);
        free(actual);

        actual = siguiente;
    }

    inicializar_lista(lista);
}

// lee un archivo y guarda sus nombres en una lista
int cargar_archivo(const char *nombre_archivo, Lista *lista) {
    FILE *archivo = fopen(nombre_archivo, "r");

    // revisa si el archivo se pudo abrir
    if (archivo == NULL) {
        printf("no se pudo abrir el archivo: %s\n", nombre_archivo);
        return 0;
    }

    char *linea = NULL;
    size_t capacidad = 0;
    ssize_t longitud;
    int resultado = 1;

    // lee el archivo linea por linea
    while ((longitud = getline(&linea, &capacidad, archivo)) != -1) {

        // elimina el salto de linea del texto
        linea[strcspn(linea, "\r\n")] = '\0';

        // guarda el nombre si la linea no esta vacia
        if (strlen(linea) > 0) {
            if (!insertar_final(lista, linea)) {
                printf("no se pudo guardar un nombre en la lista\n");
                resultado = 0;
                break;
            }
        }
    }

    // libera la memoria usada para leer las lineas
    free(linea);

    // cierra el archivo
    fclose(archivo);

    return resultado;
}

#define _POSIX_C_SOURCE 200809L
#include "arbol_chaflaneado.h"

#include <stdio.h>
#include <stdlib.h>
#include <string.h>

// inicializa una lista vacia
void lista_archivos_inicializar(ListaArchivos *lista) {
    if (!lista) return;
    lista->cabeza = NULL;
    lista->cola = NULL;
    lista->cantidad = 0;
}

// agrega un archivo al final de la lista
bool lista_archivos_agregar(ListaArchivos *lista, const char *nombre_archivo) {
    if (!lista || !nombre_archivo) return false;

    NodoArchivo *nuevo = (NodoArchivo *)malloc(sizeof(NodoArchivo));
    if (!nuevo) return false;

    nuevo->nombre_archivo = strdup(nombre_archivo);
    if (!nuevo->nombre_archivo) {
        free(nuevo);
        return false;
    }
    nuevo->siguiente = NULL;

    if (lista->cola) {
        lista->cola->siguiente = nuevo;
    } else {
        lista->cabeza = nuevo;
    }
    lista->cola = nuevo;
    lista->cantidad++;
    return true;
}

// libera la memoria utilizada por la lista
void lista_archivos_liberar(ListaArchivos *lista) {
    if (!lista) return;
    NodoArchivo *actual = lista->cabeza;
    while (actual != NULL) {
        NodoArchivo *sig = actual->siguiente;
        free(actual->nombre_archivo);
        free(actual);
        actual = sig;
    }
    lista->cabeza = NULL;
    lista->cola = NULL;
    lista->cantidad = 0;
}

// realiza una rotacion hacia la derecha
static NodoSplay *rotar_derecha(NodoSplay *x) {
    NodoSplay *y = x->izq;
    x->izq = y->der;
    y->der = x;
    return y;
}

// realiza una rotacion hacia la izquierda
static NodoSplay *rotar_izquierda(NodoSplay *x) {
    NodoSplay *y = x->der;
    x->der = y->izq;
    y->izq = x;
    return y;
}

// busca un termino y lo mueve hacia la raiz del arbol
// si no existe, mueve el ultimo nodo visitado
static NodoSplay *splay_recursivo(NodoSplay *raiz, const char *termino, bool mostrar_traza) {
    if (!raiz) return NULL;

    int cmp = strcmp(termino, raiz->termino_buscado);
    if (cmp == 0) {
        return raiz;
    }

    // busca en el subarbol izquierdo
    if (cmp < 0) {
        if (!raiz->izq) return raiz;

        int cmp_izq = strcmp(termino, raiz->izq->termino_buscado);

        if (cmp_izq < 0) {
            // caso zig-zig: dos rotaciones hacia la derecha
            raiz->izq->izq = splay_recursivo(raiz->izq->izq, termino, mostrar_traza);
            if (mostrar_traza) {
                printf("  [Splay] Rotacion Zig-Zig (derecha doble) sobre '%s'\n",
                       raiz->termino_buscado);
            }
            raiz = rotar_derecha(raiz);
        } else if (cmp_izq > 0) {
            // caso zig-zag: primero izquierda y luego derecha
            raiz->izq->der = splay_recursivo(raiz->izq->der, termino, mostrar_traza);
            if (raiz->izq->der != NULL) {
                if (mostrar_traza) {
                    printf("  [Splay] Rotacion Zig-Zag (izq en hijo '%s', der en '%s')\n",
                           raiz->izq->termino_buscado, raiz->termino_buscado);
                }
                raiz->izq = rotar_izquierda(raiz->izq);
            }
        }

        // realiza la rotacion final hacia la derecha
        if (!raiz->izq) return raiz;
        if (cmp_izq == 0 && mostrar_traza) {
            printf("  [Splay] Rotacion Zig (derecha simple) sobre '%s'\n",
                   raiz->termino_buscado);
        }
        return rotar_derecha(raiz);
    }
    // busca en el subarbol derecho
    else {
        if (!raiz->der) return raiz;

        int cmp_der = strcmp(termino, raiz->der->termino_buscado);

        if (cmp_der < 0) {
            // caso zag-zig: primero derecha y luego izquierda
            raiz->der->izq = splay_recursivo(raiz->der->izq, termino, mostrar_traza);
            if (raiz->der->izq != NULL) {
                if (mostrar_traza) {
                    printf("  [Splay] Rotacion Zag-Zig (der en hijo '%s', izq en '%s')\n",
                           raiz->der->termino_buscado, raiz->termino_buscado);
                }
                raiz->der = rotar_derecha(raiz->der);
            }
        } else if (cmp_der > 0) {
            // caso zag-zag: dos rotaciones hacia la izquierda
            raiz->der->der = splay_recursivo(raiz->der->der, termino, mostrar_traza);
            if (mostrar_traza) {
                printf("  [Splay] Rotacion Zag-Zag (izquierda doble) sobre '%s'\n",
                       raiz->termino_buscado);
            }
            raiz = rotar_izquierda(raiz);
        }

        // realiza la rotacion final hacia la izquierda
        if (!raiz->der) return raiz;
        if (cmp_der == 0 && mostrar_traza) {
            printf("  [Splay] Rotacion Zag (izquierda simple) sobre '%s'\n",
                   raiz->termino_buscado);
        }
        return rotar_izquierda(raiz);
    }
}

// crea un nuevo nodo y guarda los resultados de la busqueda
static NodoSplay *crear_nodo_splay(const char *termino, ListaArchivos *resultados) {
    NodoSplay *nuevo = (NodoSplay *)malloc(sizeof(NodoSplay));
    if (!nuevo) return NULL;

    nuevo->termino_buscado = strdup(termino);
    if (!nuevo->termino_buscado) {
        free(nuevo);
        return NULL;
    }

    if (resultados) {
        nuevo->resultados = *resultados;
        lista_archivos_inicializar(resultados);
    } else {
        lista_archivos_inicializar(&nuevo->resultados);
    }

    nuevo->accesos = 1;
    nuevo->izq = NULL;
    nuevo->der = NULL;
    return nuevo;
}

// muestra los nodos del arbol y sus resultados
static void imprimir_recursivo(const NodoSplay *nodo, int nivel) {
    if (!nodo) return;

    imprimir_recursivo(nodo->der, nivel + 1);

    for (int i = 0; i < nivel; i++) {
        printf("    ");
    }
    printf("|-- \"%s\" (libros: %zu, accesos: %zu)%s\n",
           nodo->termino_buscado,
           nodo->resultados.cantidad,
           nodo->accesos,
           (nivel == 0) ? "  <-- [RAIZ]" : "");

    imprimir_recursivo(nodo->izq, nivel + 1);
}

// libera la memoria de todos los nodos del arbol
static void destruir_recursivo(NodoSplay *nodo) {
    if (!nodo) return;
    destruir_recursivo(nodo->izq);
    destruir_recursivo(nodo->der);
    free(nodo->termino_buscado);
    lista_archivos_liberar(&nodo->resultados);
    free(nodo);
}

// crea un arbol chaflaneado vacio
ArbolChaflaneado *splay_crear(void) {
    ArbolChaflaneado *arbol = (ArbolChaflaneado *)malloc(sizeof(ArbolChaflaneado));
    if (!arbol) return NULL;
    arbol->raiz = NULL;
    arbol->total_nodos = 0;
    return arbol;
}

// busca un termino en el arbol y lo mueve hacia la raiz
NodoSplay *splay_buscar(ArbolChaflaneado *arbol, const char *termino, bool mostrar_traza) {
    if (!arbol || !termino || !arbol->raiz) return NULL;

    if (mostrar_traza) {
        printf("[Splay] Ejecutando chaflaneo para termino: \"%s\"\n", termino);
    }

    arbol->raiz = splay_recursivo(arbol->raiz, termino, mostrar_traza);

    if (arbol->raiz && strcmp(arbol->raiz->termino_buscado, termino) == 0) {
        arbol->raiz->accesos++;
        return arbol->raiz;
    }

    return NULL;
}

// inserta un termino nuevo y lo coloca en la raiz
NodoSplay *splay_insertar(ArbolChaflaneado *arbol,
                          const char *termino,
                          ListaArchivos *resultados,
                          bool mostrar_traza) {
    if (!arbol || !termino) return NULL;

    // si el arbol esta vacio, el nuevo nodo se convierte en la raiz
    if (!arbol->raiz) {
        NodoSplay *nuevo = crear_nodo_splay(termino, resultados);
        if (nuevo) {
            arbol->raiz = nuevo;
            arbol->total_nodos++;
        }
        return nuevo;
    }

    // mueve el termino mas cercano a la raiz antes de insertar
    arbol->raiz = splay_recursivo(arbol->raiz, termino, mostrar_traza);

    int cmp = strcmp(termino, arbol->raiz->termino_buscado);
    if (cmp == 0) {
        // si el termino ya existe, actualiza sus resultados
        arbol->raiz->accesos++;
        if (resultados) {
            lista_archivos_liberar(&arbol->raiz->resultados);
            arbol->raiz->resultados = *resultados;
            lista_archivos_inicializar(resultados);
        }
        return arbol->raiz;
    }

    NodoSplay *nuevo = crear_nodo_splay(termino, resultados);
    if (!nuevo) return NULL;

    if (cmp < 0) {
        nuevo->der = arbol->raiz;
        nuevo->izq = arbol->raiz->izq;
        arbol->raiz->izq = NULL;
    } else {
        nuevo->izq = arbol->raiz;
        nuevo->der = arbol->raiz->der;
        arbol->raiz->der = NULL;
    }

    arbol->raiz = nuevo;
    arbol->total_nodos++;
    return nuevo;
}

// muestra el estado actual del arbol chaflaneado
void splay_imprimir_arbol(const ArbolChaflaneado *arbol) {
    printf("\n=== Estado del Arbol Chaflaneado (Cache de Busquedas) ===\n");
    if (!arbol || !arbol->raiz) {
        printf("(Arbol vacio)\n");
        printf("=========================================================\n");
        return;
    }
    printf("Total de terminos en cache: %zu\n\n", arbol->total_nodos);
    imprimir_recursivo(arbol->raiz, 0);
    printf("=========================================================\n");
}

// libera toda la memoria utilizada por el arbol
void splay_destruir(ArbolChaflaneado *arbol) {
    if (!arbol) return;
    destruir_recursivo(arbol->raiz);
    free(arbol);
}

#include "sociedad.h"

#include <stdio.h>

// hace una copia del nombre para guardarlo en memoria
char *copiar_texto(const char *texto) {
    char *copia = malloc(strlen(texto) + 1);

    if (copia == NULL) {
        return NULL;
    }

    strcpy(copia, texto);
    return copia;
}

// agrega una persona a la lista
int agregar_persona(struct Sociedad *sociedad, const char *nombre) {
    struct Persona *nueva = malloc(sizeof(struct Persona));

    if (nueva == NULL) {
        return 0;
    }

    nueva->nombre = copiar_texto(nombre);

    if (nueva->nombre == NULL) {
        free(nueva);
        return 0;
    }

    nueva->comunaActual = NULL;
    nueva->librosLeidos = NULL;
    nueva->servicios = NULL;
    nueva->regalosDados = 0;
    nueva->regalosRecibidos = 0;
    nueva->siguiente = NULL;

    if (sociedad->personas == NULL) {
        sociedad->personas = nueva;
    } else {
        struct Persona *actual = sociedad->personas;

        while (actual->siguiente != NULL) {
            actual = actual->siguiente;
        }

        actual->siguiente = nueva;
    }

    return 1;
}

int persona_aprender_servicio(struct Persona *p, const char *servicio) {
    struct ServicioPersona *actual = p->servicios;
    struct ServicioPersona *nuevo;

    // si ya lo sabe, no lo repetimos
    while (actual != NULL) {
        if (strcmp(actual->nombre, servicio) == 0) {
            return 0;
        }
        actual = actual->siguiente;
    }

    nuevo = malloc(sizeof(struct ServicioPersona));

    if (nuevo == NULL) {
        return 0;
    }

    nuevo->nombre = copiar_texto(servicio);

    if (nuevo->nombre == NULL) {
        free(nuevo);
        return 0;
    }

    // se inserta al inicio, es lo mas simple
    nuevo->siguiente = p->servicios;
    p->servicios = nuevo;
    return 1;
}

int registrar_libro_leido(struct Persona *p, const char *nombreArchivo) {
    struct LibroLeido *actual = p->librosLeidos;
    struct LibroLeido *nuevo;

    // si ya lo leyo, no se repite
    while (actual != NULL) {
        if (strcmp(actual->nombreArchivo, nombreArchivo) == 0) {
            return 0;
        }
        actual = actual->siguiente;
    }

    nuevo = malloc(sizeof(struct LibroLeido));

    if (nuevo == NULL) {
        return 0;
    }

    nuevo->nombreArchivo = copiar_texto(nombreArchivo);

    if (nuevo->nombreArchivo == NULL) {
        free(nuevo);
        return 0;
    }

    nuevo->siguiente = p->librosLeidos;
    p->librosLeidos = nuevo;
    return 1;
}

int contar_libros_leidos(const struct Persona *p) {
    const struct LibroLeido *libro = p->librosLeidos;
    int total = 0;

    while (libro != NULL) {
        total++;
        libro = libro->siguiente;
    }

    return total;
}

int aprender_servicio_leyendo(struct Persona *p, const char *servicio) {
    struct Existencia *e;
    struct LibroLeido *libro;

    if (contar_libros_leidos(p) < 3) {
        return 0;
    }

    // devuelve 0 si ya lo sabia
    if (!persona_aprender_servicio(p, servicio)) {
        return 0;
    }

    // el stock de ese servicio crece en su comuna
    e = p->comunaActual->servicios;

    while (e != NULL) {
        if (strcmp(e->nombre, servicio) == 0) {
            e->maximo += 2;
            e->cantidad += 2;
        }
        e = e->siguiente;
    }

    // gasta los libros: para otro servicio necesita leer 3 nuevos
    while (p->librosLeidos != NULL) {
        libro = p->librosLeidos;
        p->librosLeidos = libro->siguiente;
        free(libro->nombreArchivo);
        free(libro);
    }

    return 1;
}

// calcula la reciprocidad de una comuna: 100 = dados y recibidos equilibrados
void calcular_reciprocidad_comuna(struct Sociedad *sociedad,
                                  struct Comuna *comuna) {
    int dados = 0;
    int recibidos = 0;
    struct Persona *p = sociedad->personas;

    while (p != NULL) {
        if (p->comunaActual == comuna) {
            dados += p->regalosDados;
            recibidos += p->regalosRecibidos;
        }
        p = p->siguiente;
    }

        int diferencia = dados - recibidos;
    int limite = comuna->poblacion / 10;

    if (diferencia < 0) {
        diferencia = -diferencia;
    }

    if (limite < 1) {
        limite = 1;
    }

    comuna->reciprocidad = 100.0f - 100.0f * diferencia / limite;

    if (comuna->reciprocidad < 0) {
        comuna->reciprocidad = 0;
    }
}


 // agrega una comuna y guarda su nombre en la lista doble
int agregar_comuna(struct Sociedad *sociedad, const char *nombre) {
    struct Comuna *nueva = malloc(sizeof(struct Comuna));
    struct NodoNombreDoble *nuevoNombre;

    if (nueva == NULL) {
        return 0;
    }

    nueva->nombre = copiar_texto(nombre);

    if (nueva->nombre == NULL) {
        free(nueva);
        return 0;
    }

    // inicializa los datos de la comuna
    nueva->poblacion = 0;
    nueva->bienes = NULL;
    nueva->servicios = NULL;
    nueva->necesidad = 0;
    nueva->satisfaccion = 0;
    nueva->penalizacionEmergencia = 0;
    nueva->reciprocidad = 100;
    nueva->administrada = 0;
    nueva->anterior = NULL;
    nueva->siguiente = NULL;
    nueva->biblioteca = NULL;
    
    // reserva memoria para guardar tambien el nombre
    nuevoNombre = malloc(sizeof(struct NodoNombreDoble));

    if (nuevoNombre == NULL) {
        free(nueva->nombre);
        free(nueva);
        return 0;
    }

    nuevoNombre->nombre = copiar_texto(nombre);

    if (nuevoNombre->nombre == NULL) {
        free(nuevoNombre);
        free(nueva->nombre);
        free(nueva);
        return 0;
    }

    nuevoNombre->anterior = NULL;
    nuevoNombre->siguiente = NULL;

    // agrega la comuna a la lista principal
    if (sociedad->comunas == NULL) {
        sociedad->comunas = nueva;
    } else {
        struct Comuna *actual = sociedad->comunas;

        while (actual->siguiente != NULL) {
            actual = actual->siguiente;
        }

        actual->siguiente = nueva;
        nueva->anterior = actual;
    }

    // agrega el nombre a la lista doble
    if (sociedad->nombresComunas == NULL) {
        sociedad->nombresComunas = nuevoNombre;
    } else {
        struct NodoNombreDoble *actual = sociedad->nombresComunas;

        while (actual->siguiente != NULL) {
            actual = actual->siguiente;
        }

        actual->siguiente = nuevoNombre;
        nuevoNombre->anterior = actual;
    }

    return 1;
}



// agrega un bien o servicio a las existencias de una comuna
int agregar_existencia(struct Existencia **lista, const char *nombre,
                       int maximo) {
    struct Existencia *nuevo = malloc(sizeof(struct Existencia));

    if (nuevo == NULL) {
        return 0;
    }

    nuevo->nombre = copiar_texto(nombre);

    if (nuevo->nombre == NULL) {
        free(nuevo);
        return 0;
    }

    nuevo->maximo = maximo;
    nuevo->cantidad = maximo * 4 / 5;
    nuevo->valor = 1.0f;
    nuevo->siguiente = NULL;

    if (*lista == NULL) {
        *lista = nuevo;
    } else {
        struct Existencia *actual = *lista;

        while (actual->siguiente != NULL) {
            actual = actual->siguiente;
        }

        actual->siguiente = nuevo;
    }

    return 1;
}

// crea las existencias de una comuna
int crear_existencias_comuna(struct Sociedad *sociedad,
                             struct Comuna *comuna) {
    struct NodoNombreDoble *bien = sociedad->catalogoBienes;
    struct NodoNombre *servicio = sociedad->catalogoServicios;
    int i;

    for (i = 0; i < sociedad->cantidadBienesActivos; i++) {
        if (bien == NULL) {
            return 0;
        }

        if (!agregar_existencia(&comuna->bienes, bien->nombre,
                                comuna->poblacion * 5)) {
            return 0;
        }

        bien = bien->siguiente;
    }

    for (i = 0; i < sociedad->cantidadServiciosActivos; i++) {
        if (servicio == NULL) {
            return 0;
        }

        if (!agregar_existencia(&comuna->servicios, servicio->nombre,
                                comuna->poblacion * 2)) {
            return 0;
        }

        servicio = servicio->siguiente;
    }

    return 1;
}

// calcula la necesidad y la satisfaccion de una comuna
void calcular_indices_comuna(struct Comuna *comuna) {
    struct Existencia *actual;
    int maximoTotal = 0;
    int cantidadTotal = 0;

    if (comuna == NULL) {
        return;
    }

    actual = comuna->bienes;
    while (actual != NULL) {
        maximoTotal += actual->maximo;
        cantidadTotal += actual->cantidad;
        actual = actual->siguiente;
    }

    actual = comuna->servicios;
    while (actual != NULL) {
        maximoTotal += actual->maximo;
        cantidadTotal += actual->cantidad;
        actual = actual->siguiente;
    }

    if (maximoTotal > 0) {
        comuna->necesidad =
            100.0f * (maximoTotal - cantidadTotal) / maximoTotal;
    } else {
        comuna->necesidad = 0;
    }

    if (comuna->necesidad < 0) {
        comuna->necesidad = 0;
    }

    if (comuna->necesidad >= 100) {
        comuna->necesidad = 99.9f;
    }

    comuna->satisfaccion =
    100.0f - comuna->necesidad
    - comuna->penalizacionEmergencia;

    if (comuna->satisfaccion < 0) {
        comuna->satisfaccion = 0;
    }
}


// calcula los indices de todas las comunas y mezcla la satisfaccion con la de las cercanas
void calcular_indices_sociedad(struct Sociedad *sociedad) {
    struct Comuna *actual;
    struct Comuna *vecina;
    float bienestarBase[50];
    int cantidad = 0;
    int i;
    int j;

    if (sociedad == NULL) {
        return;
    }

    actual = sociedad->comunas;

    // calcula los indices propios de cada comuna
    while (actual != NULL && cantidad < 50) {
        calcular_indices_comuna(actual);

        bienestarBase[cantidad] =
            100.0f - actual->necesidad
            - actual->penalizacionEmergencia;

        if (bienestarBase[cantidad] < 0) {
            bienestarBase[cantidad] = 0;
        }

        if (bienestarBase[cantidad] > 100) {
            bienestarBase[cantidad] = 100;
        }

        actual = actual->siguiente;
        cantidad++;
    }

    // combina el bienestar propio con el de las comunas cercanas
    actual = sociedad->comunas;

    for (i = 0; actual != NULL && i < cantidad; i++) {
        float sumaVecinas = 0;
        int cantidadVecinas = 0;
        calcular_reciprocidad_comuna(sociedad, actual);

        vecina = sociedad->comunas;

        for (j = 0; vecina != NULL && j < cantidad; j++) {
            int distancia = i - j;

            if (distancia < 0) {
                distancia = -distancia;
            }

            if (distancia >= 1 && distancia <= 2) {
                sumaVecinas += bienestarBase[j];
                cantidadVecinas++;
            }

            vecina = vecina->siguiente;
        }

        if (cantidadVecinas > 0) {
			float bienestarPropio = bienestarBase[i];
			float promedioVecinas = sumaVecinas / cantidadVecinas;

			actual->satisfaccion =
			0.6f * bienestarPropio
			+ 0.25f * promedioVecinas
			+ 0.15f * actual->reciprocidad;
			} else {
			actual->satisfaccion =
			0.85f * bienestarBase[i]
			+ 0.15f * actual->reciprocidad;
		}

        if (actual->satisfaccion < 0) {
            actual->satisfaccion = 0;
        }

        if (actual->satisfaccion > 100) {
            actual->satisfaccion = 100;
        }

        actual = actual->siguiente;
    }
}

// carga los datos iniciales de la sociedad
int inicializar_sociedad(struct Sociedad *sociedad,
                         const char *archivoPersonas,
                         const char *archivoComunas,
                         const char *archivoBienes,
                         const char *archivoServicios) {

    // deja la sociedad vacia antes de empezar
    sociedad->poblacionTotal = 0;
    sociedad->cantidadComunas = 0;
    sociedad->cantidadBienesActivos = 0;
    sociedad->cantidadServiciosActivos = 0;
    sociedad->dia = 0;
    sociedad->diasParaEmergencia = (rand() % 5) + 3;

    sociedad->personas = NULL;
    sociedad->catalogoServicios = NULL;
    sociedad->catalogoBienes = NULL;
    sociedad->nombresComunas = NULL;
    sociedad->comunas = NULL;

    // listas temporales para leer los archivos
    Lista personas;
    Lista comunas;
    Lista bienes;
    Lista servicios;

    inicializar_lista(&personas);
    inicializar_lista(&comunas);
    inicializar_lista(&bienes);
    inicializar_lista(&servicios);

    // lee los cuatro archivos
    if (!cargar_archivo(archivoPersonas, &personas) ||
        !cargar_archivo(archivoComunas, &comunas) ||
        !cargar_archivo(archivoBienes, &bienes) ||
        !cargar_archivo(archivoServicios, &servicios)) {

        liberar_lista(&personas);
        liberar_lista(&comunas);
        liberar_lista(&bienes);
        liberar_lista(&servicios);
        return 0;
    }

    // copia los nombres de las personas
    Nodo *actual = personas.inicio;

    while (actual != NULL) {
        if (!agregar_persona(sociedad, actual->nombre)) {
            liberar_lista(&personas);
            liberar_lista(&comunas);
            liberar_lista(&bienes);
            liberar_lista(&servicios);
            liberar_sociedad(sociedad);
            return 0;
        }

        sociedad->poblacionTotal++;
        actual = actual->siguiente;
    }

    // crea las comunas
    
    int comunasActivas;
    int bienesActivos;
    int serviciosActivos;

    printf("Cantidad de comunas activas (3-%d): ",
           comunas.cantidad);

    if (scanf("%d", &comunasActivas) != 1 ||
        comunasActivas < 3 ||
        comunasActivas > comunas.cantidad) {
        printf("Cantidad de comunas no valida\n");
        liberar_lista(&personas);
        liberar_lista(&comunas);
        liberar_lista(&bienes);
        liberar_lista(&servicios);
        liberar_sociedad(sociedad);
        return 0;
    }

    printf("Cantidad de bienes activos (1-%d): ",
           bienes.cantidad);

    if (scanf("%d", &bienesActivos) != 1 ||
        bienesActivos < 1 ||
        bienesActivos > bienes.cantidad) {
        printf("Cantidad de bienes no valida\n");
        liberar_lista(&personas);
        liberar_lista(&comunas);
        liberar_lista(&bienes);
        liberar_lista(&servicios);
        liberar_sociedad(sociedad);
        return 0;
    }

    printf("Cantidad de servicios activos (1-%d): ",
           servicios.cantidad);

    if (scanf("%d", &serviciosActivos) != 1 ||
        serviciosActivos < 1 ||
        serviciosActivos > servicios.cantidad) {
        printf("Cantidad de servicios no valida\n");
        liberar_lista(&personas);
        liberar_lista(&comunas);
        liberar_lista(&bienes);
        liberar_lista(&servicios);
        liberar_sociedad(sociedad);
        return 0;
    }

    sociedad->cantidadBienesActivos = bienesActivos;
    sociedad->cantidadServiciosActivos = serviciosActivos;

    // copia los bienes del catalogo
    actual = bienes.inicio;

    while (actual != NULL) {
        if (!agregar_bien(sociedad, actual->nombre)) {
            liberar_lista(&personas);
            liberar_lista(&comunas);
            liberar_lista(&bienes);
            liberar_lista(&servicios);
            liberar_sociedad(sociedad);
            return 0;
        }

        actual = actual->siguiente;
    }

    // copia los servicios del catalogo
    actual = servicios.inicio;

    while (actual != NULL) {
        if (!agregar_servicio(sociedad, actual->nombre)) {
            liberar_lista(&personas);
            liberar_lista(&comunas);
            liberar_lista(&bienes);
            liberar_lista(&servicios);
            liberar_sociedad(sociedad);
            return 0;
        }

        actual = actual->siguiente;
    }

    // crea solamente las comunas seleccionadas
    actual = comunas.inicio;

    while (actual != NULL && sociedad->cantidadComunas <
                              comunasActivas) {
        if (!agregar_comuna(sociedad, actual->nombre)) {
            liberar_lista(&personas);
            liberar_lista(&comunas);
            liberar_lista(&bienes);
            liberar_lista(&servicios);
            liberar_sociedad(sociedad);
            return 0;
        }

        sociedad->cantidadComunas++;
        actual = actual->siguiente;
    }

    // reparte la poblacion entre las comunas
    struct Comuna *comunaActual = sociedad->comunas;
    int poblacionBase = sociedad->poblacionTotal / comunasActivas;
    int personasExtra = sociedad->poblacionTotal % comunasActivas;
    int numeroComuna = 0;
    struct Persona *personaAsignar = sociedad->personas;

    while (comunaActual != NULL) {
        comunaActual->poblacion = poblacionBase;
        comunaActual->administrada = (numeroComuna < 3);

        if (numeroComuna < personasExtra) {
            comunaActual->poblacion++;
        }
        
        // asigna a esta comuna las siguientes personas de la lista
		for (int i = 0; i < comunaActual->poblacion; i++) {
		personaAsignar->comunaActual = comunaActual;
		personaAsignar = personaAsignar->siguiente;
		}

        if (!crear_existencias_comuna(sociedad, comunaActual)) {
            liberar_lista(&personas);
            liberar_lista(&comunas);
            liberar_lista(&bienes);
            liberar_lista(&servicios);
            liberar_sociedad(sociedad);
            return 0;
        }

        calcular_indices_comuna(comunaActual);

        numeroComuna++;
        comunaActual = comunaActual->siguiente;
    }


    // libera las listas temporales porque ya copiamos los datos
    liberar_lista(&personas);
    liberar_lista(&comunas);
    liberar_lista(&bienes);
    liberar_lista(&servicios);
	// asigna servicios iniciales a las personas (0, 1 o 2)
	struct Persona *personaServ = sociedad->personas;

	while (personaServ != NULL) {
		int cantidadServicios = rand() % 3;

		for (int k = 0; k < cantidadServicios; k++) {
			struct NodoNombre *servicio = sociedad->catalogoServicios;
			int posicion = rand() % sociedad->cantidadServiciosActivos;

			for (int j = 0; j < posicion; j++) {
				servicio = servicio->siguiente;
			}

			persona_aprender_servicio(personaServ, servicio->nombre);
		}	

		personaServ = personaServ->siguiente;
	}
    return 1;
}

// construye el AVL compartido y crea la biblioteca de cada comuna
int crear_bibliotecas(struct Sociedad *sociedad, const char *rutaCorpus) {
    struct Comuna *comuna;

    sociedad->indiceLibros = biblioteca_construir_indice_avl(rutaCorpus);

    if (sociedad->indiceLibros == NULL) {
        return 0;
    }

    comuna = sociedad->comunas;

    while (comuna != NULL) {
        comuna->biblioteca = biblioteca_crear_con_indice(
            comuna->nombre, sociedad->indiceLibros);

        if (comuna->biblioteca == NULL) {
            return 0;
        }

        comuna = comuna->siguiente;
    }

    return 1;
}

// libera una lista de existencias
void liberar_existencias(struct Existencia *lista) {
    struct Existencia *actual;

    while (lista != NULL) {
        actual = lista;
        lista = lista->siguiente;

        free(actual->nombre);
        free(actual);
    }
}
// disuelve una comuna: sus personas se mudan a la comuna con mas satisfaccion
// los servicios viajan con las personas y los bienes se pierden
int disolver_comuna(struct Sociedad *sociedad, struct Comuna *comuna) {
    struct Comuna *destino = NULL;
    struct Comuna *actual = sociedad->comunas;
    struct Persona *p;

    // busca la comuna destino: la de mayor satisfaccion que no sea la disuelta
    while (actual != NULL) {
        if (actual != comuna &&
            (destino == NULL || actual->satisfaccion > destino->satisfaccion)) {
            destino = actual;
        }
        actual = actual->siguiente;
    }

    if (destino == NULL) {
        return 0;
    }

    printf("La comuna %s se disuelve. Sus personas se mudan a %s.\n",
           comuna->nombre, destino->nombre);

    // muda a las personas de la comuna
    p = sociedad->personas;

    while (p != NULL) {
        if (p->comunaActual == comuna) {
            struct ServicioPersona *s = p->servicios;

            p->comunaActual = destino;
            destino->poblacion++;

            // sus servicios viajan con ella: sube el stock del destino
            while (s != NULL) {
                struct Existencia *e = destino->servicios;

                while (e != NULL) {
                    if (strcmp(e->nombre, s->nombre) == 0) {
                        e->maximo += 2;
                        e->cantidad += 2;
                    }
                    e = e->siguiente;
                }
                s = s->siguiente;
            }
        }
        p = p->siguiente;
    }

	// si era administrada, una autonoma ocupa su lugar
	if (comuna->administrada) {
		actual = sociedad->comunas;

		while (actual != NULL) {
			if (actual != comuna && actual->administrada == 0) {
				actual->administrada = 1;
				printf("La comuna %s pasa a ser administrada.\n",
					actual->nombre);
				break;
			}
			actual = actual->siguiente;
		}
}

    // saca la comuna de la lista
    if (comuna->anterior != NULL) {
        comuna->anterior->siguiente = comuna->siguiente;
    } else {
        sociedad->comunas = comuna->siguiente;
    }

    if (comuna->siguiente != NULL) {
        comuna->siguiente->anterior = comuna->anterior;
    }
	// la caché y la biblioteca de la comuna se pierden con ella
	if (comuna->biblioteca != NULL) {
		biblioteca_destruir(comuna->biblioteca);
	}
    // los bienes se pierden y se libera la comuna
    liberar_existencias(comuna->bienes);
    liberar_existencias(comuna->servicios);
    free(comuna->nombre);
    free(comuna);
    sociedad->cantidadComunas--;

    return 1;
}

// Realiza un intercambio de bienes entre dos comunas
int realizar_trueque(struct Comuna *origen,
                     struct Comuna *destino,
                     const char *bienOrigen,
                     const char *bienDestino,
                     int cantidad) {

    struct Existencia *entregaOrigen;
    struct Existencia *entregaDestino;
    struct Existencia *recibeOrigen;
    struct Existencia *recibeDestino;

    // Valida los datos del trueque
    if (origen == NULL || destino == NULL ||
        bienOrigen == NULL || bienDestino == NULL ||
        origen == destino || cantidad <= 0 ||
        strcmp(bienOrigen, bienDestino) == 0) {
        printf("Datos del trueque no validos.\n");
        return 0;
    }

    // Busca los bienes que se van a intercambiar
    entregaOrigen = buscar_bien(origen->bienes, bienOrigen);
    entregaDestino = buscar_bien(destino->bienes, bienDestino);

    recibeOrigen = buscar_bien(origen->bienes, bienDestino);
    recibeDestino = buscar_bien(destino->bienes, bienOrigen);

    // Comprueba que los cuatro bienes existan
    if (entregaOrigen == NULL || entregaDestino == NULL ||
        recibeOrigen == NULL || recibeDestino == NULL) {
        printf("No se encontraron todos los bienes necesarios.\n");
        return 0;
    }

    // Comprueba que ambas comunas tengan suficiente cantidad
    if (entregaOrigen->cantidad < cantidad ||
        entregaDestino->cantidad < cantidad) {
        printf("No hay suficientes existencias.\n");
        return 0;
    }

    // Comprueba que las comunas puedan recibir los bienes
    if (recibeOrigen->cantidad + cantidad > recibeOrigen->maximo ||
        recibeDestino->cantidad + cantidad > recibeDestino->maximo) {
        printf("El trueque supera la capacidad maxima.\n");
        return 0;
    }

    // Cada comuna entrega un bien y recibe el de la otra
    entregaOrigen->cantidad -= cantidad;
    recibeOrigen->cantidad += cantidad;

    entregaDestino->cantidad -= cantidad;
    recibeDestino->cantidad += cantidad;

    // Aumenta el valor de los bienes involucrados un 5 %
    entregaOrigen->valor *= 1.05f;
    entregaDestino->valor *= 1.05f;
    recibeOrigen->valor *= 1.05f;
    recibeDestino->valor *= 1.05f;

    // Actualiza los indices despues del intercambio
    calcular_indices_comuna(origen);
    calcular_indices_comuna(destino);

    printf("Trueque realizado entre %s y %s.\n",
           origen->nombre, destino->nombre);

    return 1;
}

int realizar_regalo(struct Persona *da, struct Persona *recibe,
                    const char *bien) {

    struct Existencia *entra;

    if (da == recibe || da->comunaActual == NULL ||
        recibe->comunaActual == NULL) {
        printf("El regalo no es valido.\n");
        return 0;
    }

    // si son de comunas distintas, se mueve una unidad del bien
    if (da->comunaActual != recibe->comunaActual) {
        sale = buscar_bien(da->comunaActual->bienes, bien);
        entra = buscar_bien(recibe->comunaActual->bienes, bien);

        if (sale == NULL || entra == NULL ||
            sale->cantidad < 1 || entra->cantidad >= entra->maximo) {
            printf("No se puede realizar el regalo.\n");
            return 0;
        }

        sale->cantidad--;
        entra->cantidad++;
    }

    da->regalosDados++;
    recibe->regalosRecibidos++;
    return 1;
}

#define _POSIX_C_SOURCE 200809L
#include "arbol_avl.h"

#include <stdlib.h>
#include <string.h>

// obtiene el numero mayor entre dos valores
static int max_int(int a, int b) {
    return (a > b) ? a : b;
}

// obtiene la altura de un nodo
static int altura_nodo(const NodoAVL *nodo) {
    return nodo ? nodo->altura : 0;
}

// calcula el balance del nodo
static int factor_balance(const NodoAVL *nodo) {
    if (!nodo) return 0;
    return altura_nodo(nodo->izq) - altura_nodo(nodo->der);
}

// crea un nuevo nodo y guarda el nombre y la ruta del archivo
static NodoAVL *crear_nodo(const char *nombre_archivo, const char *ruta_completa) {
    NodoAVL *nuevo = (NodoAVL *)malloc(sizeof(NodoAVL));
    if (!nuevo) return NULL;

    nuevo->nombre_archivo = strdup(nombre_archivo);
    nuevo->ruta_completa = strdup(ruta_completa);
    if (!nuevo->nombre_archivo || !nuevo->ruta_completa) {
        free(nuevo->nombre_archivo);
        free(nuevo->ruta_completa);
        free(nuevo);
        return NULL;
    }

    nuevo->altura = 1;
    nuevo->izq = NULL;
    nuevo->der = NULL;
    return nuevo;
}

// realiza una rotacion hacia la derecha para balancear el arbol
static NodoAVL *rotar_derecha(NodoAVL *y) {
    NodoAVL *x = y->izq;
    NodoAVL *T2 = x->der;

    x->der = y;
    y->izq = T2;

    y->altura = 1 + max_int(altura_nodo(y->izq), altura_nodo(y->der));
    x->altura = 1 + max_int(altura_nodo(x->izq), altura_nodo(x->der));

    return x;
}

// realiza una rotacion hacia la izquierda para balancear el arbol
static NodoAVL *rotar_izquierda(NodoAVL *x) {
    NodoAVL *y = x->der;
    NodoAVL *T2 = y->izq;

    y->izq = x;
    x->der = T2;

    x->altura = 1 + max_int(altura_nodo(x->izq), altura_nodo(x->der));
    y->altura = 1 + max_int(altura_nodo(y->izq), altura_nodo(y->der));

    return y;
}

// inserta un nuevo nodo y mantiene el arbol balanceado
static NodoAVL *insertar_recursivo(NodoAVL *nodo,
                                   const char *nombre_archivo,
                                   const char *ruta_completa,
                                   bool *insertado) {
    if (!nodo) {
        NodoAVL *nuevo = crear_nodo(nombre_archivo, ruta_completa);
        if (nuevo) *insertado = true;
        return nuevo;
    }

    int cmp = strcmp(nombre_archivo, nodo->nombre_archivo);
    if (cmp < 0) {
        nodo->izq = insertar_recursivo(nodo->izq, nombre_archivo, ruta_completa, insertado);
    } else if (cmp > 0) {
        nodo->der = insertar_recursivo(nodo->der, nombre_archivo, ruta_completa, insertado);
    } else {
        // si el nombre ya existe, no lo vuelve a insertar
        *insertado = false;
        return nodo;
    }

    // actualiza la altura del nodo
    nodo->altura = 1 + max_int(altura_nodo(nodo->izq), altura_nodo(nodo->der));

    // revisa si el nodo necesita balancearse
    int balance = factor_balance(nodo);

    // caso izquierda-izquierda
    if (balance > 1 && strcmp(nombre_archivo, nodo->izq->nombre_archivo) < 0) {
        return rotar_derecha(nodo);
    }

    // caso derecha-derecha
    if (balance < -1 && strcmp(nombre_archivo, nodo->der->nombre_archivo) > 0) {
        return rotar_izquierda(nodo);
    }

    // caso izquierda-derecha
    if (balance > 1 && strcmp(nombre_archivo, nodo->izq->nombre_archivo) > 0) {
        nodo->izq = rotar_izquierda(nodo->izq);
        return rotar_derecha(nodo);
    }

    // caso derecha-izquierda
    if (balance < -1 && strcmp(nombre_archivo, nodo->der->nombre_archivo) < 0) {
        nodo->der = rotar_derecha(nodo->der);
        return rotar_izquierda(nodo);
    }

    return nodo;
}

// recorre el arbol en orden y ejecuta el callback por cada nodo
static void inorder_recursivo(const NodoAVL *nodo, CallbackNodoAVL callback, void *contexto) {
    if (!nodo) return;
    inorder_recursivo(nodo->izq, callback, contexto);
    callback(nodo, contexto);
    inorder_recursivo(nodo->der, callback, contexto);
}

// libera la memoria de todos los nodos del arbol
static void destruir_recursivo(NodoAVL *nodo) {
    if (!nodo) return;
    destruir_recursivo(nodo->izq);
    destruir_recursivo(nodo->der);
    free(nodo->nombre_archivo);
    free(nodo->ruta_completa);
    free(nodo);
}

// crea e inicializa un arbol AVL vacio
ArbolAVL *avl_crear(void) {
    ArbolAVL *arbol = (ArbolAVL *)malloc(sizeof(ArbolAVL));
    if (!arbol) return NULL;
    arbol->raiz = NULL;
    arbol->total_nodos = 0;
    return arbol;
}

// inserta un libro en el arbol y actualiza la cantidad de nodos
bool avl_insertar(ArbolAVL *arbol, const char *nombre_archivo, const char *ruta_completa) {
    if (!arbol || !nombre_archivo || !ruta_completa) return false;

    bool insertado = false;
    arbol->raiz = insertar_recursivo(arbol->raiz, nombre_archivo, ruta_completa, &insertado);
    if (insertado) {
        arbol->total_nodos++;
    }
    return insertado;
}

// busca un libro por su nombre exacto
const NodoAVL *avl_buscar_exacto(const ArbolAVL *arbol, const char *nombre_archivo) {
    if (!arbol || !nombre_archivo) return NULL;

    const NodoAVL *actual = arbol->raiz;
    while (actual != NULL) {
        int cmp = strcmp(nombre_archivo, actual->nombre_archivo);
        if (cmp == 0) {
            return actual;
        } else if (cmp < 0) {
            actual = actual->izq;
        } else {
            actual = actual->der;
        }
    }
    return NULL;
}

// recorre todos los libros en orden alfabetico
void avl_recorrer_inorder(const ArbolAVL *arbol, CallbackNodoAVL callback, void *contexto) {
    if (!arbol || !callback) return;
    inorder_recursivo(arbol->raiz, callback, contexto);
}

// obtiene la altura actual del arbol
int avl_obtener_altura(const ArbolAVL *arbol) {
    if (!arbol) return 0;
    return altura_nodo(arbol->raiz);
}

// libera toda la memoria utilizada por el arbol
void avl_destruir(ArbolAVL *arbol) {
    if (!arbol) return;
    destruir_recursivo(arbol->raiz);
    free(arbol);
}

// reduce una cantidad aleatoria de una existencia
static void consumir_existencias(struct Existencia *existencias) {
    struct Existencia *actual = existencias;

    while (actual != NULL) {
        int consumo = (rand() % 5) + 1;

        if (actual->cantidad < consumo) {
            actual->cantidad = 0;
        } else {
            actual->cantidad -= consumo;
        }

        actual = actual->siguiente;
    }
}

// aplica el consumo de un dia a todas las comunas
void aplicar_consumo_diario(struct Sociedad *sociedad) {
    struct Comuna *comuna = sociedad->comunas;

    // recorre todas las comunas
    while (comuna != NULL) {
        consumir_existencias(comuna->bienes);
        consumir_existencias(comuna->servicios);

        // la penalizacion de las emergencias se recupera de a poco
        if (comuna->penalizacionEmergencia > 0) {
            comuna->penalizacionEmergencia -= 2;

            if (comuna->penalizacionEmergencia < 0) {
                comuna->penalizacionEmergencia = 0;
            }
        }

        // actualiza la necesidad y satisfaccion
        calcular_indices_comuna(comuna);

        comuna = comuna->siguiente;
    }

    // avanza al siguiente dia
    sociedad->dia++;

    // cuenta los dias que faltan para la proxima emergencia
    sociedad->diasParaEmergencia--;

    if (sociedad->diasParaEmergencia <= 0) {
        generar_emergencia(sociedad);

        // la proxima emergencia sale entre 3 y 7 dias despues
        sociedad->diasParaEmergencia = (rand() % 5) + 3;
    }
}

// busca un texto dentro de los libros
// primero revisa si el resultado ya esta guardado en cache
// si no esta, busca en los archivos y guarda el resultado
int biblioteca_buscar_texto(Biblioteca *bib,
                            const char *texto_buscado,
                            bool mostrar_traza_splay) {
    if (!bib || !bib->cache_textos || !bib->indice_libros ||
        !texto_buscado || texto_buscado[0] == '\0') {
        return -1;
    }

    char *termino_norm = duplicar_minusculas(texto_buscado);
    if (!termino_norm) return -1;

    printf("\n--- Busqueda de texto en Biblioteca (%s): \"%s\" ---\n",
           bib->nombre_comuna, termino_norm);

    // revisa primero si el texto esta en el arbol chaflaneado
    NodoSplay *en_cache = splay_buscar(bib->cache_textos, termino_norm, mostrar_traza_splay);
    if (en_cache != NULL) {
        bib->cache_hits++;
        printf("[CACHE HIT] Resultado recuperado desde el Arbol Chaflaneado (accesos: %zu).\n",
               en_cache->accesos);
        imprimir_lista_resultados(&en_cache->resultados, termino_norm);
        int total = (int)en_cache->resultados.cantidad;
        free(termino_norm);
        return total;
    }

    // si no esta en cache, busca en los libros del disco
    bib->cache_misses++;
    printf("[CACHE MISS] Buscando \"%s\" en %zu libros en disco...\n",
           termino_norm, bib->indice_libros->total_nodos);

    ListaArchivos encontrados;
    lista_archivos_inicializar(&encontrados);

    ContextoBusquedaDisco ctx = {
        .texto_normalizado = termino_norm,
        .resultados = &encontrados,
        .archivos_revisados = 0
    };

    avl_recorrer_inorder(bib->indice_libros, callback_buscar_en_libro, &ctx);

    int total_hallados = (int)encontrados.cantidad;
    imprimir_lista_resultados(&encontrados, termino_norm);

    // guarda el resultado en el arbol chaflaneado
    splay_insertar(bib->cache_textos, termino_norm, &encontrados, mostrar_traza_splay);

    free(termino_norm);
    return total_hallados;
}

#define _POSIX_C_SOURCE 200809L
#include "archivos.h"

#include <ctype.h>
#include <dirent.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <sys/stat.h>

#define BUFFER_LINEA 4096
#define MAX_RUTA 1024

// verifica si el archivo termina en .txt
static bool es_archivo_txt(const char *nombre) {
    if (!nombre) return false;
    size_t len = strlen(nombre);
    if (len < 4) return false;

    const char *ext = nombre + (len - 4);
    return (tolower((unsigned char)ext[0]) == '.' &&
            tolower((unsigned char)ext[1]) == 't' &&
            tolower((unsigned char)ext[2]) == 'x' &&
            tolower((unsigned char)ext[3]) == 't');
}

// busca un texto dentro de otro ignorando mayusculas y minusculas
static const char *strstr_insensible(const char *pajar, const char *aguja) {
    if (!*aguja) return pajar;

    for (; *pajar; pajar++) {
        if (tolower((unsigned char)*pajar) == tolower((unsigned char)*aguja)) {
            const char *h = pajar + 1;
            const char *n = aguja + 1;
            while (*h && *n &&
                   tolower((unsigned char)*h) == tolower((unsigned char)*n)) {
                h++;
                n++;
            }
            if (!*n) return pajar;
        }
    }
    return NULL;
}

int archivos_recorrer_directorio(const char *ruta_directorio,
                                 CallbackArchivoTxt callback,
                                 void *contexto) {
    if (!ruta_directorio || !callback) return -1;

    DIR *dir = opendir(ruta_directorio);
    if (!dir) {
        perror("Error al abrir directorio de biblioteca");
        return -1;
    }

    int total_encontrados = 0;
    struct dirent *entrada;
    char ruta_completa[MAX_RUTA];

    while ((entrada = readdir(dir)) != NULL) {
        // ignora la carpeta actual y la carpeta anterior
        if (strcmp(entrada->d_name, ".") == 0 ||
            strcmp(entrada->d_name, "..") == 0) {
            continue;
        }

        // construye la ruta completa del archivo o carpeta
        size_t len_dir = strlen(ruta_directorio);
        int escritos;
        if (len_dir > 0 && ruta_directorio[len_dir - 1] == '/') {
            escritos = snprintf(ruta_completa, sizeof(ruta_completa), "%s%s",
                                ruta_directorio, entrada->d_name);
        } else {
            escritos = snprintf(ruta_completa, sizeof(ruta_completa), "%s/%s",
                                ruta_directorio, entrada->d_name);
        }

        // verifica que la ruta no supere el tamaño del buffer
        if (escritos < 0 || (size_t)escritos >= sizeof(ruta_completa)) {
            fprintf(stderr, "Advertencia: ruta demasiado larga omitida: %s/%s\n",
                    ruta_directorio, entrada->d_name);
            continue;
        }

        struct stat info;
        if (stat(ruta_completa, &info) != 0) {
            continue;
        }

        // si encuentra una carpeta, la recorre tambien
        if (S_ISDIR(info.st_mode)) {
            int sub_total = archivos_recorrer_directorio(ruta_completa, callback, contexto);
            if (sub_total > 0) {
                total_encontrados += sub_total;
            }
        // si es un archivo .txt, ejecuta el callback
        } else if (S_ISREG(info.st_mode) && es_archivo_txt(entrada->d_name)) {
            callback(entrada->d_name, ruta_completa, contexto);
            total_encontrados++;
        }
    }

    closedir(dir);
    return total_encontrados;
}

bool archivos_contiene_texto(const char *ruta_archivo,
                           const char *texto_buscado,
                           bool ignorar_mayusculas) {
    if (!ruta_archivo || !texto_buscado || texto_buscado[0] == '\0') {
        return false;
    }

    FILE *archivo = fopen(ruta_archivo, "r");
    if (!archivo) {
        return false;
    }

    char linea[BUFFER_LINEA];
    bool encontrado = false;

    // lee el archivo linea por linea hasta encontrar el texto
    while (fgets(linea, sizeof(linea), archivo) != NULL) {
        const char *match = ignorar_mayusculas
                                ? strstr_insensible(linea, texto_buscado)
                                : strstr(linea, texto_buscado);
        if (match != NULL) {
            encontrado = true;
            break; // ya encontro el texto, no necesita seguir leyendo
        }
    }

    fclose(archivo);
    return encontrado;
}

bool archivos_mostrar_fragmento(const char *ruta_archivo, size_t max_lineas) {
    if (!ruta_archivo || max_lineas == 0) return false;

    FILE *archivo = fopen(ruta_archivo, "r");
    if (!archivo) {
        perror("No se pudo abrir el libro");
        return false;
    }

    char linea[BUFFER_LINEA];
    size_t contador = 0;

    // muestra las primeras lineas del archivo
    printf("\n--- Inicio de extracto (%s) ---\n", ruta_archivo);
    while (contador < max_lineas && fgets(linea, sizeof(linea), archivo) != NULL) {
        printf("%s", linea);
        contador++;
    }
    printf("\n--- Fin de extracto ---\n");

    fclose(archivo);
    return true;
}

## Descripción del proyecto

Sociedad Solarpunk es un proyecto desarrollado en lenguaje C que simula el funcionamiento de una sociedad formada por diferentes comunas. Cada comuna cuenta con una población y existencias de bienes y servicios que pueden variar a lo largo de la simulación.

El programa permite administrar las existencias de las comunas, realizar trueques de bienes y servicios entre ellas y simular el consumo diario de los recursos disponibles. Además, contempla la generación de emergencias que provocan pérdidas de bienes y servicios y afectan el bienestar de la comuna donde ocurren y de algunas comunas cercanas.

Durante la simulación se actualizan los índices de necesidad y satisfacción de las comunas, lo que permite observar cómo cambian sus condiciones a lo largo del tiempo. Al finalizar, se presenta un resumen del estado de la sociedad.

El proyecto permite poner en práctica conceptos de programación en C, estructuras de datos, manejo de memoria y organización del código en diferentes archivos.

#include <stdio.h>
#include <stdlib.h>
#include "emergencias.h"
#include "sociedad.h"

// colores para mostrar los mensajes
#define ROJO "\033[31m"
#define AMARILLO "\033[33m"
#define CIAN "\033[36m"
#define VERDE "\033[32m"
#define NEGRITA "\033[1m"
#define RESET "\033[0m"

// quita el 25 % de la cantidad de cada bien o servicio de una lista
static void perder_cuarto(struct Existencia *lista) {
    struct Existencia *actual = lista;
    int reduccion;
    int huboPerdidas = 0;

    while (actual != NULL) {
        reduccion = actual->cantidad / 4;

        // solo muestra lo que de verdad se perdio
        if (reduccion > 0) {
            actual->cantidad -= reduccion;

            printf("  - %-30s " ROJO "-%d unidades\n" RESET,
                   actual->nombre, reduccion);

            huboPerdidas = 1;
        }

        actual = actual->siguiente;
    }

    if (!huboPerdidas) {
        printf("  " VERDE "No hubo perdidas en esta categoria.\n" RESET);
    }
}

// genera una emergencia en una comuna aleatoria
void generar_emergencia(struct Sociedad *sociedad) {
    struct Comuna *comuna;
    struct Comuna *afectada;
    int cantidadComunas = 0;
    int posicion;
    int i;

    if (sociedad == NULL) {
        return;
    }

    comuna = sociedad->comunas;

    // cuenta las comunas disponibles
    while (comuna != NULL) {
        cantidadComunas++;
        comuna = comuna->siguiente;
    }

    // revisa si hay comunas disponibles
    if (cantidadComunas == 0) {
        return;
    }

    // selecciona una comuna al azar
    posicion = rand() % cantidadComunas;
    afectada = sociedad->comunas;

    for (i = 0; i < posicion; i++) {
        afectada = afectada->siguiente;
    }

    // muestra el aviso principal de la emergencia
    printf("\n");
    printf(AMARILLO NEGRITA);
    printf("==================================================\n");
    printf("                 ALERTA DEL SISTEMA\n");
    printf("==================================================\n");
    printf(RESET);

    printf(ROJO NEGRITA);
    printf("\n  ¡EMERGENCIA EN LA COMUNA %s!\n", afectada->nombre);
    printf(RESET);

    printf("\nSe ha registrado una emergencia que afecta a esta comuna.\n");
    printf("Se aplicaran perdidas en sus bienes y servicios.\n");

    // muestra las perdidas de bienes
    printf("\n" CIAN NEGRITA "  PERDIDAS DE BIENES\n" RESET);
    printf("  ------------------------------------------------\n");
    perder_cuarto(afectada->bienes);

    // muestra las perdidas de servicios
    printf("\n" CIAN NEGRITA "  PERDIDAS DE SERVICIOS\n" RESET);
    printf("  ------------------------------------------------\n");
    perder_cuarto(afectada->servicios);

    // la satisfaccion de la comuna afectada baja directo
    afectada->penalizacionEmergencia += 20;

    // actualiza la necesidad y satisfaccion de la comuna afectada
    calcular_indices_comuna(afectada);

    // reduce la satisfaccion de las comunas cercanas
    afectar_comunas_cercanas(afectada);

    printf("\n" AMARILLO NEGRITA);
    printf("  La emergencia tambien puede afectar a las comunas cercanas.\n");
    printf("==================================================\n");
    printf(RESET);
}

// baja la satisfaccion de una comuna cercana (si existe)
static void penalizar_cercana(struct Comuna *cercana) {
    if (cercana == NULL) {
        return;
    }

    cercana->penalizacionEmergencia += 10;

    // actualiza la necesidad y satisfaccion
    calcular_indices_comuna(cercana);

    printf(AMARILLO "  [!] La comuna %s tambien fue afectada.\n" RESET,
           cercana->nombre);
}

// las cercanas son la anterior, la siguiente y la siguiente de la siguiente
void afectar_comunas_cercanas(struct Comuna *afectada) {
    if (afectada == NULL) {
        return;
    }

    penalizar_cercana(afectada->anterior);
    penalizar_cercana(afectada->siguiente);

    if (afectada->siguiente != NULL) {
        penalizar_cercana(afectada->siguiente->siguiente);
    }
}
